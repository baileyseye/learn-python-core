## Переменные и числа

```python
name = "Анна"          # str
age = 27               # int
price = 199.9          # float
active = True          # bool
unknown = None         # NoneType

a, b = 1, 2
a, b = b, a            # обмен без временной переменной
first, *rest = [1, 2, 3]

17 // 5                # 3
17 % 5                 # 2
2 ** 10                # 1024
round(2.675, 2)        # 2.67 — float приблизителен
```

## Строки

```python
text = "  Learn Python  "

text.replace("Python", "API")
```

## Коллекции

```python
items = [1, 2, 3]
items.append(4)
items[0] = 10

point = (3, 7)               # tuple, неизменяемый
x, y = point

user = {"name": "Анна", "age": 27}
user.get("email", "—")
user["city"] = "Москва"
for key, value in user.items():
    print(key, value)

tags = {"python", "api"}
tags | {"sql"}               # объединение
tags & {"python"}            # пересечение

squares = [n ** 2 for n in range(5)]
evens = {n for n in range(10) if n % 2 == 0}
lengths = {word: len(word) for word in ["a", "bb"]}
```

```python
    return f"{greeting}, {name}!"


def total(*numbers: int) -> int:
```

## Классы

```python
from dataclasses import dataclass, field


class Account:
    def __init__(self, owner: str, balance: int = 0):
        self.owner = owner
        self.balance = balance

    def deposit(self, amount: int) -> None:
        self.balance += amount


@dataclass
class Task:
    title: str
    done: bool = False
    tags: list[str] = field(default_factory=list)
```

```python
try:
```
